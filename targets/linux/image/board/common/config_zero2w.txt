# The Morning Newspaper display: Raspberry Pi firmware settings (config.txt).
# Based on Buildroot's board/raspberrypi sample.

start_file=start.elf
fixup_file=fixup.dat

kernel=zImage

# Graphics memory as in Buildroot's sample (the panel does not use the GPU).
gpu_mem_256=100
gpu_mem_512=100
gpu_mem_1024=100

# Serial console on the header (ttyAMA0), Bluetooth on the mini UART.
dtoverlay=miniuart-bt

# Pimoroni Inky Impression 13.3": SPI on, and SPI keeps its hands off the
# kernel chip select pins (the panel's chip selects are GPIO 26 and 16,
# moved by the program). From Pimoroni's inky library setup.
dtparam=spi=on
dtoverlay=spi0-0cs
